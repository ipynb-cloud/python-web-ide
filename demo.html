Quick Setup

            

                
<!-- 1. Include the library script -->
<script src="https://python-web-ide.pages.dev/python-web-ide.js"></script>

<!-- 2. Create an empty container -->
<div id="my-python-editor"></div>

<!-- 3. Initialize the editor -->
<script>
  const editor = new PythonEditor({
      container: '#my-python-editor',
      id: 'question_123',             
      height: '500px',                
      starterCode: 'print("Hello")',  
      onChange: (code) => {           
          console.log("Updated!", code);
      }
  });
</script>
